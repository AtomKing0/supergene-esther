Q1. 대시보드 진입 시 빅쿼리가 실시간으로 호출되나?

아니요 — 진입 시 빅쿼리 호출 0건임
대시보드는 정적 HTML(GitHub Pages)이고, 데이터는 빌드 시점에 HTML 안에 JSON으로 박제돼 있음.
브라우저는 그 JSON을 렌더만 함. 화면 숫자 = 마지막 크론 빌드 스냅샷(최신 정시분, 최대 ~1시간 지연). 새로고침해도 BQ 안 감 — 다음 크론이 파일을 교체해야 갱신됨.

RAW daily() — 일별 DAU/NRU/판수 + 로딩 p50
WITH ev AS (
      SELECT log_date, player_id, event,
        CAST(JSON_EXTRACT_SCALAR(data,'$.time') AS INT64) AS t,
        CAST(JSON_EXTRACT_SCALAR(data,'$.request_time') AS INT64) AS rt,
        MAX(IF(event='1000_LOGIN_COMPLETE' AND logincount_total=1,1,0))
          OVER(PARTITION BY log_date,player_id) AS is_nru,
        MAX(IF(event='1000_LOGIN_COMPLETE',1,0))
          OVER(PARTITION BY log_date,player_id) AS has_login
      FROM `game-log-359704.raw.cozy_tiles_prod`
      WHERE log_date BETWEEN DATE_SUB(CURRENT_DATE('Asia/Seoul'),INTERVAL 3 DAY) AND DATE_SUB(CURRENT_DATE('Asia/Seoul'),INTERVAL 1 DAY)
        AND is_dev IS DISTINCT FROM TRUE AND player_id>0
    ),
    pl AS (  -- 유저-일당 로딩값(ld)
      SELECT log_date, player_id, MAX(is_nru) is_nru, MAX(has_login) has_login, CASE WHEN MIN(IF(event='2100_GAMEPLAY_START', rt, NULL)) IS NOT NULL AND MIN(IF(event='1010_LOAD_COMPLETE' AND t IS NOT NULL, rt - t, NULL)) IS NOT NULL AND MIN(IF(event='2100_GAMEPLAY_START', rt, NULL)) - (MIN(IF(event='1010_LOAD_COMPLETE' AND t IS NOT NULL, rt - t, NULL))) >= 0 THEN MIN(IF(event='2100_GAMEPLAY_START', rt, NULL)) - (MIN(IF(event='1010_LOAD_COMPLETE' AND t IS NOT NULL, rt - t, NULL))) END AS ld
      FROM ev GROUP BY log_date, player_id
    ),
    ldd AS (  -- 일별 로딩 중앙값(세그별)
      SELECT log_date,
        APPROX_QUANTILES(ld,100)[OFFSET(50)]                              AS load_md,
        APPROX_QUANTILES(IF(is_nru=1,ld,NULL),100)[OFFSET(50)]            AS nru_load_md,
        APPROX_QUANTILES(IF(is_nru=0 AND has_login=1,ld,NULL),100)[OFFSET(50)] AS ru_load_md
      FROM pl GROUP BY log_date
    )
    SELECT m.log_date,
      m.nru_u, m.nru_g, m.nru_rvf, m.nru_start, m.nru_touch, m.nru_finish,
      m.ru_u, m.ru_g, m.ru_rvf, m.nru_fc, m.ru_fc,
      m.dau_u, m.dau_g, m.dau_fc, m.dau_rvf,
      ldd.load_md, ldd.nru_load_md, ldd.ru_load_md
    FROM (
      SELECT log_date,
        COUNT(DISTINCT IF(is_nru=1,player_id,NULL))               AS nru_u,
        COUNTIF(event='2100_GAMEPLAY_START' AND is_nru=1)         AS nru_g,
        COUNTIF(event LIKE '%RV_FINISH%' AND is_nru=1)            AS nru_rvf,
        COUNT(DISTINCT IF(event='2100_GAMEPLAY_START' AND is_nru=1,player_id,NULL)) AS nru_start,
        COUNT(DISTINCT IF(event='2115_GAMEPLAY_FIRST_TOUCH' AND is_nru=1,player_id,NULL))            AS nru_touch,
        COUNT(DISTINCT IF(event='2300_GAMEPLAY_FINISH' AND is_nru=1,player_id,NULL)) AS nru_finish,
        COUNT(DISTINCT IF(is_nru=0 AND has_login=1,player_id,NULL))               AS ru_u,
        COUNTIF(event='2100_GAMEPLAY_START' AND is_nru=0 AND has_login=1)         AS ru_g,
        COUNTIF(event LIKE '%RV_FINISH%' AND is_nru=0 AND has_login=1)            AS ru_rvf,
        COUNTIF(event='2300_GAMEPLAY_FINISH' AND is_nru=1)                   AS nru_fc,
        COUNTIF(event='2300_GAMEPLAY_FINISH' AND is_nru=0 AND has_login=1)   AS ru_fc,
        COUNT(DISTINCT IF(has_login=1,player_id,NULL))                       AS dau_u,
        COUNTIF(event='2100_GAMEPLAY_START' AND has_login=1)                 AS dau_g,
        COUNTIF(event='2300_GAMEPLAY_FINISH' AND has_login=1)                AS dau_fc,
        COUNTIF(event LIKE '%RV_FINISH%' AND has_login=1)                    AS dau_rvf
      FROM ev GROUP BY 1
    ) m LEFT JOIN ldd USING(log_date) ORDER BY m.log_date
RAW session_metrics — 세션수·세션타임
WITH u AS (
      SELECT log_date, player_id,
        MAX(IF(event='1000_LOGIN_COMPLETE' AND logincount_total=1,1,0)) AS is_nru,
        MAX(IF(event='1000_LOGIN_COMPLETE',logincount_today,NULL))      AS sessions,
        MAX(IF(event LIKE '%PLAYTIME%',
               CAST(JSON_EXTRACT_SCALAR(data,'$.session_time') AS INT64),NULL)) AS max_pt
      FROM `game-log-359704.raw.cozy_tiles_prod`
      WHERE log_date BETWEEN DATE('2026-01-04') AND DATE_ADD(DATE('2026-01-04'),INTERVAL 20 DAY)
        AND is_dev IS DISTINCT FROM TRUE AND player_id>0
      GROUP BY 1,2
    )
    SELECT
      AVG(IF(is_nru=1,sessions,NULL))                              AS nru_sess,
      AVG(IF(is_nru=1,max_pt,NULL))                                AS nru_pt,
      AVG(IF(is_nru=0,sessions,NULL))                              AS ru_sess,
      AVG(IF(is_nru=0,max_pt,NULL))                                AS ru_pt,
      AVG(sessions)                                                AS dau_sess,
      AVG(max_pt)                                                  AS dau_pt
    FROM u
RAW retention_matrix — 리텐션 코호트 조인
WITH nru AS (
      SELECT DISTINCT log_date AS d, player_id
      FROM `game-log-359704.raw.cozy_tiles_prod`
      WHERE log_date BETWEEN DATE('2026-01-04') AND DATE_ADD(DATE('2026-01-04'),INTERVAL 20 DAY)
        AND event='1000_LOGIN_COMPLETE' AND logincount_total=1
        AND is_dev IS DISTINCT FROM TRUE AND player_id>0
    ),
    act AS (
      SELECT DISTINCT log_date AS d, player_id
      FROM `game-log-359704.raw.cozy_tiles_prod`
      WHERE log_date BETWEEN DATE_ADD(DATE('2026-01-04'),INTERVAL 1 DAY)
                         AND DATE_ADD(DATE_ADD(DATE('2026-01-04'),INTERVAL 20 DAY),INTERVAL 1 DAY)
        AND event='1000_LOGIN_COMPLETE'  -- 복귀=로그인 기준 (루커 정합, 7/12 확정)
        AND is_dev IS DISTINCT FROM TRUE AND player_id>0
    )
    SELECT n.d AS cohort_date,
      COUNT(DISTINCT n.player_id)                                  AS cohort_n,
      COUNT(DISTINCT IF(a.player_id IS NOT NULL,n.player_id,NULL)) AS ret_n
    FROM nru n
    LEFT JOIN act a
      ON a.player_id=n.player_id AND a.d=DATE_ADD(n.d,INTERVAL 1 DAY)
    GROUP BY 1 ORDER BY 1

      SELECT log_date, stat_name, player_type, SUM(player_count) pc, SUM(log_count) lc

      GROUP BY 1,2,3)
